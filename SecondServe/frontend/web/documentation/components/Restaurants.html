    document.addEventListener("DOMContentLoaded", function() {
      // Initialize map
      const map = L.map('sponsor-map').setView([39.8283, -98.5795], 4);

      // Add OpenStreetMap layer
      L.tileLayer('https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png', {
        attribution: '© OpenStreetMap contributors'
      }).addTo(map);

      // Example sponsor locations
      const sponsors = [
        { name: "Joe's Deli", type: "Restaurant", coords: [40.7128, -74.0060], availability: "Yes" },
        { name: "FreshMart", type: "Grocery Store", coords: [34.0522, -118.2437], availability: "Limited" },
        { name: "Tasty Eats", type: "Restaurant", coords: [41.8781, -87.6298], availability: "Yes" }
      ];

      // Add markers
      sponsors.forEach(sponsor => {
        const marker = L.marker(sponsor.coords).addTo(map);
        marker.bindPopup(`<b>${sponsor.name}</b><br>${sponsor.type}<br>Availability: ${sponsor.availability}`);
        marker.on('mouseover', function() { this.openPopup(); });
        marker.on('mouseout', function() { this.closePopup(); });
      });
    });
  </script>
