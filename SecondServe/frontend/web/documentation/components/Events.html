Events Component

Displays the events page showcasing community events, upcoming activities, and calendar information. This component serves as a container for events-related content and provides routing capabilities for navigation within the events section.

Features:

  • Standalone component with minimal dependencies
  • Router integration for navigation
  • Clean separation of template and styles

This component is typically rendered when navigating to the '/events' route and displays event listings, calendar views, and community activities.

    document.addEventListener("DOMContentLoaded", function () {
      // --- Events Data (you can later pull from DB or API)
      const eventsData = [
        { title: "Food Drive - Dallas", date: "2025-11-03" },
        { title: "Volunteer Orientation", date: "2025-11-05" },
        { title: "Community Garden Event", date: "2025-11-06" },
        { title: "Thanksgiving Meal Prep", date: "2025-11-09" },
        { title: "Holiday Food Distribution", date: "2025-11-25" },
      ];

      // --- Monthly Calendar ---
      const calendarEl = document.getElementById("monthly-calendar");
      const calendar = new FullCalendar.Calendar(calendarEl, {
        initialView: 'dayGridMonth',
        events: eventsData,
      });
      calendar.render();

      // --- Weekly Events List ---
      const weeklyContainer = document.getElementById("weekly-events");
      const today = new Date();
      const startOfWeek = new Date(today);
      startOfWeek.setDate(today.getDate() - today.getDay()); // Sunday

      const endOfWeek = new Date(startOfWeek);
      endOfWeek.setDate(startOfWeek.getDate() + 6); // Saturday

      const weekEvents = eventsData.filter(event => {
        const eventDate = new Date(event.date);
        return eventDate >= startOfWeek && eventDate <= endOfWeek;
      });

      if (weekEvents.length === 0) {
        weeklyContainer.innerHTML = `<p>No events scheduled for this week.</p>`;
      } else {
        weeklyContainer.innerHTML = weekEvents.map(ev => `
          <div class="event-card">
            <h3>${ev.title}</h3>
            <p>${new Date(ev.date).toLocaleDateString("en-US", { weekday: "long", month: "long", day: "numeric" })}</p>
          </div>
        `).join("");
      }
    });
  </script>
