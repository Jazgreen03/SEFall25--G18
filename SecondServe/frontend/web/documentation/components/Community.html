    document.addEventListener("DOMContentLoaded", function() {
      const calendarPopup = document.getElementById("calendar-popup");
      const viewCalendarBtn = document.getElementById("view-calendar");
      const closeCalendarBtn = document.getElementById("close-calendar");

      // Show calendar
      viewCalendarBtn.addEventListener("click", function() {
        calendarPopup.style.display = "block";

        // Initialize calendar once
        if (!window.calendarInitialized) {
          const calendarEl = document.getElementById("calendar");
          const calendar = new FullCalendar.Calendar(calendarEl, {
            initialView: 'dayGridMonth',
            events: [
              { title: 'Food Drive - Dallas', date: '2025-11-15' },
              { title: 'Volunteer Meetup - Chicago', date: '2025-11-20' },
              { title: 'Holiday Food Distribution', date: '2025-12-01' }
            ]
          });
          calendar.render();
          window.calendarInitialized = true;
        }
      });

      // Close calendar
      closeCalendarBtn.addEventListener("click", function() {
        calendarPopup.style.display = "none";
      });
    });
  </script>
